3つのエンドポイント

テキスト取得・リスト取得・スクリーンショットの3種類をシンプルな REST API で提供

GET / POST /scrape/text ページのテキストコンテンツを取得
パラメータ
名前型必須説明
urlstring必須取得対象のURL
selectorstring任意CSSセレクタ（指定時はその要素のみ）
wait_forstring任意待機するCSSセレクタ
timeoutnumber任意タイムアウト秒数（デフォルト30）
レスポンス例
JSON 
{
  "success": true,
  "url": "https://example.com",
  "text": "ページのテキスト内容...",
  "selector": null,
  "elapsed_ms": 1842
}
GET / POST /scrape/list 繰り返し要素をリスト形式で取得
パラメータ
名前型必須説明
urlstring必須取得対象のURL
selectorstring必須繰り返し要素のCSSセレクタ
fieldsobject任意抽出フィールド定義（key: セレクタ）
limitnumber任意最大取得件数（デフォルト50）
レスポンス例
JSON 
{
  "success": true,
  "count": 20,
  "items": [
    { "title": "商品名A", "price": "¥1,980" },
    { "title": "商品名B", "price": "¥2,480" }
  ],
  "elapsed_ms": 2310
}
GET / POST /scrape/screenshot ページのスクリーンショットを取得
パラメータ
名前型必須説明
urlstring必須スクリーンショット対象のURL
full_pageboolean任意フルページキャプチャ（デフォルト false）
widthnumber任意ビューポート幅（デフォルト 1280）
formatstring任意png / jpeg（デフォルト png）
レスポンス例
JSON 
{
  "success": true,
  "url": "https://example.com",
  "image_base64": "iVBORw0KGgoAAAA...",
  "format": "png",
  "width": 1280,
  "height": 900,
  "elapsed_ms": 3210
}

JavaScript (Fetch) 
// テキスト取得の例
const response = await fetch(
  'https://playwright-scraping-api.p.rapidapi.com/scrape/text',
  {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'X-RapidAPI-Key': 'YOUR_RAPIDAPI_KEY',
      'X-RapidAPI-Host': 'playwright-scraping-api.p.rapidapi.com',
    },
    body: JSON.stringify({
      url: 'https://example.com/products',
      selector: '.product-title',
      wait_for: '.product-list',
    }),
  }
);

const data = await response.json();
console.log(data.text);
// => "商品タイトルのテキスト..."

// リスト取得の例
const listRes = await fetch(
  'https://playwright-scraping-api.p.rapidapi.com/scrape/list',
  {
    method: 'POST',
    headers: { /* 上と同じ headers */ },
    body: JSON.stringify({
      url: 'https://example.com/products',
      selector: '.product-item',
      fields: { title: '.name', price: '.price' },
      limit: 20,
    }),
  }
);
const list = await listRes.json();
console.log(list.items);
Node.js (axios) 
const axios = require('axios');

async function scrapeText(url, selector) {
  const { data } = await axios.post(
    'https://playwright-scraping-api.p.rapidapi.com/scrape/text',
    { url, selector },
    {
      headers: {
        'X-RapidAPI-Key': process.env.RAPIDAPI_KEY,
        'X-RapidAPI-Host': 'playwright-scraping-api.p.rapidapi.com',
      },
    }
  );
  return data.text;
}

// スクリーンショット取得
async function takeScreenshot(url) {
  const { data } = await axios.post(
    'https://playwright-scraping-api.p.rapidapi.com/scrape/screenshot',
    { url, full_page: true, format: 'jpeg' },
    { headers: { /* ... */ } }
  );
  // base64 -> Buffer
  const buf = Buffer.from(data.image_base64, 'base64');
  require('fs').writeFileSync('screenshot.jpg', buf);
}
Python (requests) 
import requests, os, base64

HEADERS = {
    "X-RapidAPI-Key": os.environ["RAPIDAPI_KEY"],
    "X-RapidAPI-Host": "playwright-scraping-api.p.rapidapi.com",
}
BASE = "https://playwright-scraping-api.p.rapidapi.com"

# テキスト取得
def scrape_text(url, selector=None):
    res = requests.post(
        f"{BASE}/scrape/text",
        json={"url": url, "selector": selector},
        headers=HEADERS,
    )
    return res.json()["text"]

# リスト取得
def scrape_list(url, selector, fields=None, limit=50):
    res = requests.post(
        f"{BASE}/scrape/list",
        json={"url": url, "selector": selector,
              "fields": fields, "limit": limit},
        headers=HEADERS,
    )
    return res.json()["items"]

# スクリーンショット保存
def save_screenshot(url, path="screenshot.png"):
    res = requests.post(
        f"{BASE}/scrape/screenshot",
        json={"url": url, "full_page": True},
        headers=HEADERS,
    )
    img = base64.b64decode(res.json()["image_base64"])
    with open(path, "wb") as f:
        f.write(img)
